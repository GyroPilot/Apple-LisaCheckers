unit CkrNet;

{ REV C  18-Oct-2026  Network READs of 128 bytes, not 256 (LisaEm: a     }
{                      256-byte READ took 9-12 s and could time out).    }

{ REV B  18-Oct-2026  LisaCheckers 0.2: play over the network.  New unit. }
{                      NOT yet compiled on a Lisa.                        }

{ CKR-NET.TEXT -- talking to the checkers table server (Tim's Pi, public }
{ through Tailscale Funnel, so no player opens a port) through an RS-232 }
{ FujiNet on Serial B, with LisaFuji's FnBus unit linked as it is.  Every }
{ request is one HTTPS GET:                                               }
{    N1:HTTPS://lisa-gw1.tailc2f141.ts.net/checkers<path>                 }
{          ?table=<table>&player=<name>                                   }
{ <path> is /tables, /state, /move/11-15, /resign, /draw, /leave or       }
{ /again.  The answer is plain text, one fact a line ("status play",     }
{ "board bbbbbbbbbbbb........wwwwwwwwwwww", ...); this unit reads it as   }
{ it arrives, so no reply buffer is needed (the "legal" line, which can  }
{ be long, is not wanted: CkrRules knows the moves).                      }

INTERFACE

uses (*$U UnitStd.obj *)   UnitStd,
     (*$U UnitHz.obj *)    UnitHz,
     (*$U Storage.obj *)   Storage,
     (*$U QuickDraw.obj *) QuickDraw,
     (*$U SysCall.obj *)   SysCall,
     (*$U CKR/FNBUS.OBJ *) FnBus,
     (*$U CKR/RULES.OBJ *) CkrRules;

const
   cnIdle  = 15000;           { ms a fetch waits with nothing arriving }
   cnTbMax = 6;
   cnLobby = 0;               { table status }
   cnPlay  = 1;
   cnOver  = 2;

type
   CnNmStr = string[8];
   CnLnStr = string[60];
   CnByte  = -128..127;

var
   cnTable  : CnNmStr;
   cnName   : CnNmStr;
   cnBaud   : longint;
   cnVol    : E_Name;         { the tool's volume, from CnLoad (phase 2) }
   cnWhy    : CnLnStr;        { why the last request failed }
   cnRc     : integer;
   cnEcode  : integer;
   cnBuf    : array [0..64] of CnByte;   { the name file }

   { the last state }
   cnGot    : boolean;        { a CK1 state came back }
   cnStat   : integer;        { cnLobby, cnPlay, cnOver }
   cnYou    : integer;        { ckBlack, ckWhite, 0 = watching }
   cnTurn   : integer;        { ckBlack, ckWhite, 0 }
   cnBlack  : CnNmStr;
   cnWhite  : CnNmStr;
   cnBoard  : CkBoard;
   cnHaveBd : boolean;
   cnLast   : CnLnStr;        { "22x15x8", or "-" }
   cnMoves  : integer;
   cnDraw   : integer;        { who has offered a draw, 0 nobody }
   cnResult : CnLnStr;
   cnPrompt : CnLnStr;
   cnErr    : CnLnStr;        { the server's ERROR line, '' if none }

   { the last table list }
   cnTbN    : integer;
   cnTbId   : array [1..cnTbMax] of CnNmStr;
   cnTbName : array [1..cnTbMax] of string[20];
   cnTbPl   : array [1..cnTbMax] of string[4];
   cnTbSt   : array [1..cnTbMax] of string[6];

procedure CnInit;
function  CnGet   (var path : Str255) : boolean;
procedure CnLastMv (var mv : CkMove);
function  CnLoad  (var vol : E_Name) : boolean;
function  CnSave  (var vol : E_Name) : boolean;
procedure CnSetSp  (nb : longint; var msg : Str255);


IMPLEMENTATION

{$S CKRNet}

procedure CnInit;
begin
   cnTable := '';
   cnName := 'lisa';
   cnBaud := 9600;
   cnVol := '';
   cnWhy := '';
   cnRc := 0;
   cnEcode := 0;
   cnGot := false;
   cnStat := cnLobby;
   cnYou := 0;
   cnTurn := 0;
   cnBlack := '';
   cnWhite := '';
   cnHaveBd := false;
   cnLast := '-';
   cnMoves := 0;
   cnDraw := 0;
   cnResult := '';
   cnPrompt := '';
   cnErr := '';
   cnTbN := 0;
   FnInit
end;

function CnSide (var v : Str255) : integer;
begin
   if v = 'black' then
      CnSide := ckBlack
   else if v = 'white' then
      CnSide := ckWhite
   else
      CnSide := 0
end;

procedure CnCut (var s : Str255; n : integer);
begin
   if length(s) > n then s[0] := chr(n)
end;

procedure CnTbLine (var v : Str255);
{ One /tables line after its T: id|name|n/2|status }
var parts : array [1..4] of Str255;
    k, i  : integer;
    one   : string[1];
begin
   for k := 1 to 4 do parts[k] := '';
   k := 1;
   one := ' ';
   for i := 1 to length(v) do
      if v[i] = '|' then begin
         if k < 4 then k := k + 1
      end
      else begin
         one[1] := v[i];
         parts[k] := concat(parts[k], one)
      end;
   if (k = 4) and (cnTbN < cnTbMax) then begin
      cnTbN := cnTbN + 1;
      CnCut(parts[1], 8);
      cnTbId[cnTbN] := parts[1];
      CnCut(parts[2], 20);
      cnTbName[cnTbN] := parts[2];
      CnCut(parts[3], 4);
      cnTbPl[cnTbN] := parts[3];
      CnCut(parts[4], 6);
      cnTbSt[cnTbN] := parts[4]
   end
end;

procedure CnLine (var s : Str255);
{ One line of a reply: "key value". }
var p, i, n : integer;
    key     : Str255;
    v       : Str255;
    c       : char;
begin
   p := pos(' ', s);
   if p = 0 then begin
      key := s;
      v := ''
   end
   else begin
      key := copy(s, 1, p - 1);
      v := '';
      if p < length(s) then
         v := copy(s, p + 1, length(s) - p)
   end;
   if key = 'CK1' then
      cnGot := true
   else if (length(key) = 1) and (key[1] = 'T') then
      CnTbLine(v)
   else if (length(s) > 2) and (s[1] = 'T') and (s[2] = '|') then begin
      v := copy(s, 3, length(s) - 2);
      CnTbLine(v)
   end
   else if key = 'ERROR' then begin
      CnCut(v, 60);
      cnErr := v
   end
   else if key = 'status' then begin
      if v = 'play' then cnStat := cnPlay
      else if v = 'over' then cnStat := cnOver
      else cnStat := cnLobby
   end
   else if key = 'you' then
      cnYou := CnSide(v)
   else if key = 'turn' then
      cnTurn := CnSide(v)
   else if key = 'black' then begin
      CnCut(v, 8);
      cnBlack := v
   end
   else if key = 'white' then begin
      CnCut(v, 8);
      cnWhite := v
   end
   else if key = 'board' then begin
      if length(v) >= 32 then begin
         for i := 1 to 32 do begin
            c := v[i];
            if c = 'b' then n := 1
            else if c = 'B' then n := 2
            else if c = 'w' then n := -1
            else if c = 'W' then n := -2
            else n := 0;
            cnBoard[i] := n
         end;
         cnHaveBd := true
      end
   end
   else if key = 'last' then begin
      CnCut(v, 60);
      cnLast := v
   end
   else if key = 'moves' then begin
      n := 0;
      for i := 1 to length(v) do
         if (v[i] >= '0') and (v[i] <= '9') then n := n * 10 + ord(v[i]) - 48;
      cnMoves := n
   end
   else if key = 'draw' then
      cnDraw := CnSide(v)
   else if key = 'result' then begin
      CnCut(v, 60);
      cnResult := v
   end
   else if key = 'prompt' then begin
      CnCut(v, 60);
      cnPrompt := v
   end
end;

function CnGet {var path : Str255} : boolean;
{ One request; the reply's lines go through CnLine as they arrive.     }
{ FALSE: cnWhy says why.  The read loop is LisaFuji's NwGet.            }
var url        : Str255;
    rc, k, i   : integer;
    b          : integer;
    idle       : longint;
    t0         : longint;
    clk        : boolean;
    done, fail : boolean;
    cur        : Str255;
    one        : string[1];
    part       : Str255;
    mbWhy      : Str255;      { REV N: why the mailbox was not used }
    old        : longint;
begin
   CnGet := false;
   cnRc := 0;
   cnWhy := '';
   cnErr := '';
   cnGot := false;
   if path = '/tables' then
      cnTbN := 0;
   url := concat('N1:HTTPS://lisa-gw1.tailc2f141.ts.net/checkers', path);
   url := concat(url, concat('?table=', cnTable));
   url := concat(url, concat('&player=', cnName));
   if path = '/tables' then
      url := concat(url, '&dev=1');
   if not fnFound then begin            { phase 2: which speed answers?  Once per launch. }
      old := cnBaud;
      if FnFind(cnBaud) = fnOK then
         if cnBaud <> old then
            if length(cnVol) > 0 then
               if CnSave(cnVol) then ;
   end;
   if fnVia = fnViaMb then           { REV N: in words a player can act on }
      mbWhy := ''
   else begin
      FnNum(fnMbEc, part);
      if fnMbEc = 0 then
         mbWhy := concat('mailbox on ', concat(fnVol, ' did not answer: ESPFuji?'))
      else
         mbWhy := concat('no mailbox on ', concat(fnVol, concat(' (ec ', concat(part, '): put it beside ESPFuji'))))
   end;
   if not FnPortUp(cnBaud) then begin
      cnWhy := 'Serial B would not open (is another tool using it?)';
      exit(CnGet)
   end;
   rc := FnNClos;
   rc := FnNOpen(url, 4, 0);
   if rc <> fnOK then begin
      cnRc := rc;
      if length(mbWhy) = 0 then
         cnWhy := 'the FujiNet did not take the request'
      else begin
         if length(mbWhy) > 58 then mbWhy := copy(mbWhy, 1, 58);   { REV N: copy() past the end gives '' (notes 93) }
         cnWhy := mbWhy
      end;
      rc := FnNClos;
      FnPortDn;
      exit(CnGet)
   end;
   if fnStat <> 1 then begin
      FnNum(fnStat, part);
      if fnStat = 136 then
         cnWhy := 'the checkers server sent nothing back'
      else if fnStat = 170 then
         cnWhy := 'the checkers server was not found (status 170): WiFi?'
      else
         cnWhy := concat('the checkers server refused it, status ', part);
      rc := FnNClos;
      FnPortDn;
      exit(CnGet)
   end;
   idle := 0;
   FnTStart(t0, clk);
   done := false;
   fail := false;
   cur := '';
   one := ' ';
   while not (done or fail) do begin
      if fnWait = 0 then begin
         rc := FnNStat;
         if rc <> fnOK then
            fail := true
         else if fnWait = 0 then begin
            if fnStat = 136 then
               done := true
            else if (fnConn = 0) and (fnStat <> 1) then
               done := true
            else begin
               FnNap(100);
               idle := idle + 100;
               if FnTOver(t0, clk, idle, cnIdle) then begin
                  fail := true;
                  rc := fnNoRep
               end
            end
         end
      end
      else begin
         k := 128;                        { REV C: 256-byte reads timed out in LisaEm (pitfall 29) }
         if fnWait < k then
            k := fnWait;
         rc := FnNRead(k);
         if rc <> fnOK then
            fail := true
         else begin
            idle := 0;
            FnTStart(t0, clk);
            for i := 0 to fnPlLen - 1 do begin
               b := FnPlB(i);
               if b = 10 then begin
                  CnLine(cur);
                  cur := ''
               end
               else if (b >= 32) and (b < 127) and (length(cur) < 120) then begin
                  one[1] := chr(b);
                  cur := concat(cur, one)
               end
            end
         end
      end
   end;
   if length(cur) > 0 then
      CnLine(cur);
   k := FnNClos;
   FnPortDn;
   if fail then begin
      cnRc := rc;
      cnWhy := 'the answer from the checkers server stopped part-way';
      exit(CnGet)
   end;
   CnGet := true
end;

procedure CnLastMv {var mv : CkMove};
{ The last move ("22x15x8" or "11-15") as a CkMove, to mark it on the  }
{ board.  nHop = 0 when there is none.                                  }
var i, n : integer;
    have : boolean;
begin
   mv.nHop := -1;
   mv.nCap := 0;
   n := 0;
   have := false;
   for i := 1 to length(cnLast) + 1 do begin
      if i <= length(cnLast) then
         if (cnLast[i] >= '0') and (cnLast[i] <= '9') then begin
            n := n * 10 + ord(cnLast[i]) - 48;
            have := true
         end;
      if (i > length(cnLast)) or (cnLast[i] = '-') or (cnLast[i] = 'x') then
         if have then begin
            if (mv.nHop < ckMaxHop) and (n >= 1) and (n <= 32) then begin
               mv.nHop := mv.nHop + 1;
               mv.hop[mv.nHop] := n
            end;
            n := 0;
            have := false
         end
   end;
   if mv.nHop < 1 then
      mv.nHop := 0
end;

function CnLoad {var vol : E_Name} : boolean;
var path     : Pathname;
    refnum   : integer;
    got      : longint;
    ec       : integer;
    schedErr : integer;
    i, b     : integer;
    fld      : integer;
    f        : array [0..2] of string[12];
    one      : string[1];
begin
   CnLoad := false;
   cnVol := vol;                        { phase 2: for a save from CnGet or CnSetSp }
   path := concat(vol, 'CKRNAME.DATA');
   Sched_Class(schedErr, false);
   OPEN(ec, path, refnum, [DRead]);
   Sched_Class(schedErr, true);
   if ec > 0 then
      exit(CnLoad);
   got := 0;
   Sched_Class(schedErr, false);
   READ_DATA(ec, refnum, ord4(@cnBuf), 64, got, sequential, 0);
   CLOSE_OBJECT(ec, refnum);
   Sched_Class(schedErr, true);
   for i := 0 to 2 do f[i] := '';
   fld := 0;
   one := ' ';
   for i := 0 to got - 1 do begin
      b := cnBuf[i];
      if b < 0 then b := b + 256;
      if (b = 124) and (fld < 2) then
         fld := fld + 1
      else if (b > 32) and (b <= 126) and (length(f[fld]) < 12) then begin
         one[1] := chr(b);
         f[fld] := concat(f[fld], one)
      end
   end;
   if (f[0] <> 'CN1') or (fld < 2) or (length(f[1]) = 0) then
      exit(CnLoad);
   if length(f[1]) > 8 then f[1][0] := chr(8);
   cnName := f[1];
   if f[2] = '19200' then cnBaud := 19200 else cnBaud := 9600;
   CnLoad := true
end;

function CnSave {var vol : E_Name} : boolean;
{ TwinClock's way: 890 from MAKE_FILE = it is already there. }
var path     : Pathname;
    refnum   : integer;
    ec       : integer;
    schedErr : integer;
    wrote    : longint;
    line     : Str255;
    i, n     : integer;
begin
   CnSave := false;
   line := concat('CN1|', concat(cnName, '|'));
   if cnBaud = 19200 then
      line := concat(line, '19200')
   else
      line := concat(line, '9600');
   n := length(line);
   for i := 1 to n do
      cnBuf[i - 1] := ord(line[i]);
   cnBuf[n] := 13;
   n := n + 1;
   path := concat(vol, 'CKRNAME.DATA');
   Sched_Class(schedErr, false);
   MAKE_FILE(cnEcode, path, 0);
   Sched_Class(schedErr, true);
   if (cnEcode > 0) and (cnEcode <> 890) then
      exit(CnSave);
   Sched_Class(schedErr, false);
   OPEN(cnEcode, path, refnum, [DWrite]);
   Sched_Class(schedErr, true);
   if cnEcode > 0 then
      exit(CnSave);
   Sched_Class(schedErr, false);
   WRITE_DATA(cnEcode, refnum, ord4(@cnBuf), n, wrote, sequential, 0);
   if (cnEcode <= 0) and (wrote = n) then begin
      TRUNCATE(ec, refnum);
      CnSave := true
   end;
   CLOSE_OBJECT(ec, refnum);
   Sched_Class(schedErr, true);
   if cnEcode < 0 then cnEcode := 0
end;


procedure CnSetSp {nb : longint; var msg : Str255};
{ Phase 2 (22-Oct): a speed chosen on the menu.  The FujiNet is set too, }
{ through its web page (FnSetBd), restarted and followed; the choice is  }
{ kept in the settings file.  The message says how it ended.             }
var rc, stage : integer;
begin
   if nb = cnBaud then begin
      FnSpdMsg(fnOK, 0, nb, msg);
      exit(CnSetSp)
   end;
   rc := FnSetBd(cnBaud, nb, stage);
   if stage >= 4 then
      cnBaud := nb;                    { the FujiNet was told, or is there already }
   if length(cnVol) > 0 then
      if CnSave(cnVol) then ;
   FnSpdMsg(rc, stage, cnBaud, msg)
end;
