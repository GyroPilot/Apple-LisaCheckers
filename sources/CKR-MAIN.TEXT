procedure NumStr (n : longint; var s : Str255);
var tmp : string[1];
    neg : boolean;
begin
   neg := n < 0;
   if neg then n := -n;
   s := '';
   tmp := '0';
   repeat
      tmp[1] := chr(ord('0') + n mod 10);
      s := concat(tmp, s);
      n := n div 10
   until n = 0;
   if neg then s := concat('-', s)
end;

{ Make the window current (Reversi's way). }

end;

function ScrSq (sr, sc : integer) : integer;
{ The square at screen row sr, column sc (0..7, row 0 at the top).  Your }
{ side is always at the bottom: as black the board is turned round.     }
begin
   if human = ckBlack then
      ScrSq := CkSq(7 - sr, 7 - sc)
   else
      ScrSq := CkSq(sr, sc)
end;

procedure SqScr (sq : integer; var sr, sc : integer);
{ Where square sq is on the screen. }
begin
   sr := CkRow(sq);
   sc := CkCol(sq);
   if human = ckBlack then begin
      sr := 7 - sr;
      sc := 7 - sc
   end
end;

procedure CellRect (sr, sc : integer; var rc : Rect);
var x, y : integer;
begin
   x := boardX + sc * cellPx;
   y := boardY + sr * cellPx;
   SetRect(rc, x, y, x + cellPx, y + cellPx)
end;

function Picked (sq : integer) : boolean;
{ sq is the picked piece or a square clicked after it. }
var i : integer;
begin
   Picked := false;
   for i := 0 to nPath do
      if path[i] = sq then Picked := true
end;

function Matches (var mv : CkMove) : boolean;
{ mv starts at path[0] and passes the squares clicked after it, in order. }
var i, j  : integer;
    ok    : boolean;
    found : boolean;
begin
   ok := false;
   if nPath >= 0 then
      ok := mv.hop[0] = path[0];
   j := 0;
   i := 1;
   while ok and (i <= nPath) do begin
      found := false;                      { no short-circuit OR on the Lisa: }
      while (not found) and (j < mv.nHop) do begin   { never index past nHop }
         j := j + 1;
         if mv.hop[j] = path[i] then found := true
      end;
      if not found then ok := false;
      i := i + 1
   end;
   Matches := ok
end;

function CanGo (sq : integer) : boolean;
{ A move that fits the clicks so far can land on sq next (for the dots). }
var i, j : integer;
    go   : boolean;
begin
   CanGo := false;
   if nPath < 0 then
      exit(CanGo);
   for i := 1 to nLegal do
      if Matches(legal[i]) then begin
         go := false;
         for j := 1 to legal[i].nHop do
            if legal[i].hop[j] = sq then go := true;
         if go then CanGo := true
      end
end;

procedure Shrink (var d : Rect; n : integer);
{ d made smaller by n on every side, with SetRect (proven on the Lisa). }
begin
   SetRect(d, d.left + n, d.top + n, d.right - n, d.bottom - n)
end;

procedure DrawSq (sr, sc : integer);
{ One square: light, or gray with its piece; the picked piece marked in  }
{ its corners, dots where it can go, the Lisa's last move marked top     }
{ left, the cursor as a double frame, the number when # is on.           }
var sq, p, i : integer;
    rc, d    : Rect;
    s        : Str255;
    mark     : boolean;
begin
   CellRect(sr, sc, rc);
   SetRect(d, rc.left + 1, rc.top + 1, rc.right, rc.bottom);
   sq := ScrSq(sr, sc);
   if sq = 0 then begin
      EraseRect(d);
      exit(DrawSq)
   end;
   PenNormal;
   PenPat(ckrGray);
   PaintRect(d);
   PenNormal;
   p := ckBd[sq];
   SetRect(d, rc.left + 4, rc.top + 4, rc.right - 3, rc.bottom - 3);
   if p > 0 then begin                     { black: solid }
      EraseOval(d);
      Shrink(d, 1);
      PaintOval(d);
      if p = 2 then begin                  { a king: a white ring inside }
         Shrink(d, 6);
         EraseOval(d);
         Shrink(d, 3);
         PaintOval(d)
      end
   end
   else if p < 0 then begin                { white: outlined }
      EraseOval(d);
      FrameOval(d);
      if p = -2 then begin                 { a king: a black ring inside }
         Shrink(d, 7);
         PaintOval(d);
         Shrink(d, 3);
         EraseOval(d)
      end
   end
   else if showDots and (not thinking) and (not gameOver) and CanGo(sq) then begin
      SetRect(d, rc.left + 11, rc.top + 11, rc.left + 24, rc.top + 24);
      EraseOval(d);
      Shrink(d, 3);
      PaintOval(d)
   end;
   if (nPath >= 0) and Picked(sq) then begin
      SetRect(d, rc.left + 1, rc.top + 1, rc.left + 6, rc.top + 6);
      PaintRect(d);
      SetRect(d, rc.right - 5, rc.top + 1, rc.right, rc.top + 6);
      PaintRect(d);
      SetRect(d, rc.left + 1, rc.bottom - 5, rc.left + 6, rc.bottom);
      PaintRect(d);
      SetRect(d, rc.right - 5, rc.bottom - 5, rc.right, rc.bottom);
      PaintRect(d)
   end;
   mark := false;
   if haveLisa then
      for i := 0 to lastLisa.nHop do
         if lastLisa.hop[i] = sq then mark := true;
   if mark then begin
      SetRect(d, rc.left + 2, rc.top + 2, rc.left + 9, rc.top + 9);
      EraseRect(d);
      SetRect(d, rc.left + 3, rc.top + 3, rc.left + 8, rc.top + 8);
      PaintRect(d)
   end;
   if showNums then begin
      NumStr(sq, s);
      SetRect(d, rc.right - 15, rc.bottom - 12, rc.right - 1, rc.bottom - 1);
      EraseRect(d);
      MoveTo(rc.right - 14, rc.bottom - 2);
      DrawString(s)
   end;
   if (sr = cursorR) and (sc = cursorC) then begin
      SetRect(d, rc.left + 1, rc.top + 1, rc.right, rc.bottom);
      FrameRect(d);
      SetRect(d, rc.left + 2, rc.top + 2, rc.right - 1, rc.bottom - 1);
      FrameRect(d)
   end
end;

procedure DrawBoard;
var i, x, y : integer;
    sr, sc  : integer;
    rc      : Rect;
begin
   SetRect(rc, 0, 0, panelX, winH);
   EraseRect(rc);
   for i := 0 to 8 do begin
      x := boardX + i * cellPx;
      MoveTo(x, boardY);
      LineTo(x, boardY + boardPx);
      y := boardY + i * cellPx;
      MoveTo(boardX, y);
      LineTo(boardX + boardPx, y)
   end;
   for sr := 0 to 7 do
      for sc := 0 to 7 do
         DrawSq(sr, sc)
end;

procedure StartTurn;
{ Your turn: your legal moves, nothing picked; repaint. }
begin
   CkGen(ckBd, human, legal, nLegal);
   nPath := -1;
   DrawShow
end;

procedure FinishGame (who : integer);
{ who: the winner (ckBlack or ckWhite), 0 for a draw. }
begin
   gameOver := true;
   thinking := false;
   nLegal := 0;
   nPath := -1;
   if who = human then
      status := 'Game over: you win.'
   else if who = lisaC then
      status := 'Game over: Lisa wins.'
   else
      status := 'Game over: a draw.';
   DrawShow;
   if who = human then
      NoteAlert(allAlerts, alrtWin)
   else if who = lisaC then
      NoteAlert(allAlerts, alrtLose)
   else
      NoteAlert(allAlerts, alrtDraw)
end;

procedure Played (var mv : CkMove);
{ After a move: the count toward a draw (only kings moving, nothing     }
{ taken), then whose turn it is.                                         }
begin
   if (mv.nCap > 0) or (abs(ckBd[mv.hop[mv.nHop]]) = 1) then
      ckQuiet := 0
   else
      ckQuiet := ckQuiet + 1;
   ckSide := -ckSide
end;

procedure LisaTurn;
{ The Lisa moves; then it is your turn, or the game is over. }
var n  : integer;
    mv : CkMove;
begin
   thinking := true;
   status := 'Lisa is thinking...';
   SyncPort;
   DrawPanel;
   n := CkThink(lvDepth[level], lvBudget[level], mv);
   nodes := ckNodes;
   thinking := false;
   if n = 0 then begin
      FinishGame(human);                   { the Lisa cannot move }
      exit(LisaTurn)
   end;
   CkDo(ckBd, mv);
   Played(mv);
   lastLisa := mv;
   haveLisa := true;
   CkGen(ckBd, human, legal, nLegal);
   if nLegal = 0 then begin
      FinishGame(lisaC);
      exit(LisaTurn)
   end;
   if ckQuiet >= drawPly then begin
      FinishGame(0);
      exit(LisaTurn)
   end;
   if legal[1].nCap > 0 then
      status := 'Your move.  You must jump.'
   else
      status := 'Your move.';
   StartTurn
end;

{------------------------------------------------------------------------}
{ The network game (REV B): a table on the checkers server.  The server  }
{ keeps the board and checks every move; the Lisa shows it, lets you     }
{ pick your move with CkrRules' list, and looks again every pollGap      }
{ seconds from the idle loop.                                             }
{------------------------------------------------------------------------}

function NowSecs : longint;
var ok : boolean;
begin
   NowSecs := FnSecs(ok)
end;

procedure NetSync;
{ The server's state onto the board. }
begin
   if cnHaveBd then
      ckBd := cnBoard;
   if cnYou <> 0 then
      human := cnYou
   else
      human := ckBlack;                  { watching: black at the bottom }
   lisaC := -human;
   if cnTurn <> 0 then
      ckSide := cnTurn;
   CnLastMv(lastLisa);
   haveLisa := lastLisa.nHop > 0;
   haveYou := false;
   gameOver := cnStat = cnOver;
   thinking := false;
   if cnMoves <> lastMvCnt then begin    { a new position: nothing picked }
      nPath := -1;
      lastMvCnt := cnMoves
   end;
   if (cnStat = cnPlay) and (cnYou <> 0) and (cnTurn = cnYou) then begin
      CkGen(ckBd, human, legal, nLegal);
      if (nLegal > 0) and (legal[1].nCap > 0) then
         status := 'Your move.  You must jump.'
      else
         status := 'Your move.'
   end
   else begin
      nLegal := 0;
      nPath := -1;
      if cnStat = cnOver then
         status := concat(cnResult, '.')
      else
         status := cnPrompt
   end
end;

function NetAsk (path : Str255) : boolean;
{ One request; the window is redrawn with what came back. }
var ok : boolean;
begin
   status := 'Talking to the checkers server ...';
   if myWindow <> nil then begin
      SyncPort;
      DrawPanel
   end;
   ok := CnGet(path);
   lastPoll := NowSecs;
   if ok and cnGot then
      NetSync
   else if not ok then
      status := concat('No answer: ', cnWhy)
   else if path <> '/tables' then
      status := 'The server sent something unexpected.';
   if cnErr <> '' then
      status := concat('Server: ', cnErr);
   NetAsk := ok;
   DrawShow
end;

procedure NetSend (var mv : CkMove);
{ Your move, to the server; the board comes back from it. }
var s : Str255;
begin
   CkText(mv, s);
   thinking := true;
   if NetAsk(concat('/move/', s)) then ;
end;

procedure YouMove (var mv : CkMove);
{ Your move is complete: make it, then the Lisa's reply. }
begin
   if netOn then begin                       { REV B: the server checks it and answers }
      NetSend(mv);
      exit(YouMove)
   end;
   if nHist < maxHist then begin
      nHist := nHist + 1;
      hist[nHist].bd := ckBd;
      hist[nHist].quiet := ckQuiet
   end;
   CkDo(ckBd, mv);
   Played(mv);
   lastYou := mv;
   haveYou := true;
   started := true;
   nPath := -1;
   nLegal := 0;
   if ckQuiet >= drawPly then begin
      FinishGame(0);
      exit(YouMove)
   end;
   status := 'Lisa is thinking...';
   thinking := true;
   DrawShow;
   LisaTurn
end;

procedure PickSq (sq : integer);
{ A click (or RETURN) on square sq: pick a piece, or say where it goes. }
var i, n, last : integer;
    own        : boolean;
    one        : integer;
begin
   if (myWindow = nil) or gameOver or thinking or (sq = 0) then
      exit(PickSq);
   own := ckBd[sq] * human > 0;
   if own then begin                        { pick this piece }
      n := 0;
      for i := 1 to nLegal do
         if legal[i].hop[0] = sq then n := n + 1;
      if n = 0 then begin
         if legal[1].nCap > 0 then
            status := 'That piece cannot move: you must jump.'
         else
            status := 'That piece cannot move.';
         nPath := -1
      end
      else begin
         nPath := 0;
         path[0] := sq;
         status := 'Now click where it goes.'
      end;
      DrawShow;
      exit(PickSq)
   end;
   if nPath < 0 then begin
      status := 'First click one of your pieces.';
      SyncPort;
      DrawPanel;
      exit(PickSq)
   end;
   if nPath < maxPath then begin
      nPath := nPath + 1;
      path[nPath] := sq
   end;
   n := 0;                                  { moves that fit the clicks }
   last := 0;                               { and end where the last click was }
   one := 0;
   for i := 1 to nLegal do
      if Matches(legal[i]) then begin
         n := n + 1;
         if legal[i].hop[legal[i].nHop] = sq then begin
            last := last + 1;
            one := i
         end
      end;
   if n = 0 then begin
      nPath := nPath - 1;
      status := 'It cannot go there.';
      DrawShow;
      exit(PickSq)
   end;
   if (n = 1) and (last = 1) then begin
      YouMove(legal[one]);
      exit(PickSq)
   end;
   if last >= 1 then                        { two routes end here: ask the way }
      status := 'Two ways to get there: click each landing square.'
   else
      status := 'Keep going: click the next landing square.';
   DrawShow
end;

procedure NewGame;
begin
   CkStart;
   nHist := 0;
   gameOver := false;
   started := false;
   thinking := false;
   haveYou := false;
   haveLisa := false;
   nodes := 0;
   nPath := -1;
   cursorR := 5;
   cursorC := 2;
   if human = ckBlack then begin
      status := 'Your move.  Black moves first.';
      StartTurn
   end
   else begin
      status := 'Lisa is black and opens.';
      nLegal := 0;
      DrawShow;
      LisaTurn
   end
end;

procedure UndoMove;
{ Back to the position before your last move (the Lisa's reply goes too). }
begin
   if (myWindow = nil) or thinking then
      exit(UndoMove);
   if netOn then begin
      status := 'No take-backs in a network game.';
      SyncPort;
      DrawPanel;
      exit(UndoMove)
   end;
   if nHist = 0 then begin
      status := 'Nothing to take back.';
      SyncPort;
      DrawPanel;
      exit(UndoMove)
   end;
   ckBd := hist[nHist].bd;
   ckQuiet := hist[nHist].quiet;
   nHist := nHist - 1;
   ckSide := human;
   gameOver := false;
   haveYou := false;
   haveLisa := false;
   nodes := 0;
   started := nHist > 0;
   status := 'Took back your last move and the reply.';
   StartTurn
end;

procedure AskNewGame;
begin
   if netOn then begin
      status := 'In a network game: Another Game, or Leave (Network menu).';
      SyncPort;
      DrawPanel;
      exit(AskNewGame)
   end;
   if started and (not gameOver) then begin
      if CautionAlert(allAlerts, alrtNewG) then NewGame
   end
   else
      NewGame
end;

procedure SwitchSides;
begin
   if netOn then begin
      status := 'The server gives the colours (they swap each game).';
      SyncPort;
      DrawPanel;
      exit(SwitchSides)
   end;
   if started and (not gameOver) then
      if not CautionAlert(allAlerts, alrtNewG) then
         exit(SwitchSides);
   human := -human;
   lisaC := -human;
   NewGame
end;

procedure MoveCursor (dr, dc : integer);
var oldR, oldC : integer;
begin
   if myWindow = nil then
      exit(MoveCursor);
   oldR := cursorR;
   oldC := cursorC;
   cursorR := cursorR + dr;
   cursorC := cursorC + dc;
   if cursorR < 0 then cursorR := 0;
   if cursorR > 7 then cursorR := 7;
   if cursorC < 0 then cursorC := 0;
   if cursorC > 7 then cursorC := 7;
   SyncPort;
   DrawSq(oldR, oldC);
   DrawSq(cursorR, cursorC)
end;

procedure ClickSq (where : Point);
{ A click on the board.  where is window-local (the LisaFuji lesson). }
var sr, sc : integer;
begin
   if myWindow = nil then
      exit(ClickSq);
   if (where.h < boardX) or (where.v < boardY) then
      exit(ClickSq);
   if (where.h >= boardX + boardPx) or (where.v >= boardY + boardPx) then
      exit(ClickSq);
   sc := (where.h - boardX) div cellPx;
   sr := (where.v - boardY) div cellPx;
   cursorR := sr;
   cursorC := sc;
   PickSq(ScrSq(sr, sc))
end;

end;

   lastMvCnt := -1;

end;

      if NetAsk('/leave') then ;

procedure NameKey (ch : char);
var tmp : string[1];
begin
   if (ch = chr(13)) or (ch = chr(3)) then begin
      if length(nameTxt) > 0 then begin
         cnName := nameTxt;
         if CnSave(myVol) then ;
      end;
      viewMd := vwTables;
      DrawShow
   end
   else if ch = chr(8) then begin
      if length(nameTxt) > 0 then
         nameTxt[0] := chr(length(nameTxt) - 1);
      DrawShow
   end
   else if ch = chr(27) then begin
      viewMd := vwTables;
      DrawShow
   end
   else if length(nameTxt) < 8 then begin
      if (ch >= 'A') and (ch <= 'Z') then ch := chr(ord(ch) + 32);
      if ((ch >= 'a') and (ch <= 'z')) or ((ch >= '0') and (ch <= '9')) then begin
         tmp := ' ';
         tmp[1] := ch;
         nameTxt := concat(nameTxt, tmp);
         DrawShow
      end
   end
end;

procedure IdlePoll;
{ From the idle loop: look at the server every pollGap seconds at a     }
{ table (every 15 while it is your move, to notice a resignation).      }
var gap : longint;
begin
   if (myWindow = nil) or (not netOn) or (viewMd <> vwBoard) or thinking then
      exit(IdlePoll);
   gap := pollGap;
   if nLegal > 0 then gap := 15;
   if NowSecs - lastPoll >= gap then
      if NetAsk('/state') then ;
end;

procedure MenuCommand (menu, item : integer);
var err : integer;
begin
   case menu of

      mFile :
         case item of
            miAsideAll : DoFilingCmd(cmdClosAll);
            miAside    : DoFilingCmd(cmdClose);
            miPutAway  :
               begin
                  TellFiler(err, docClosd, docPutBack, myWindow);
                  myWindow := nil
               end;
         end;

      mGame :
         case item of
            miNew   : AskNewGame;
            miUndo  : UndoMove;
            miDots  : ToggleDots;
            miNums  : ToggleNums;
            miSides : SwitchSides;
         end;

      mLevel :
         SetLevel(item);

      mNet :
         case item of
            miNetPlay  : if netOn then LeaveTable else LookTables;
            miNetLeave : LeaveTable;
            miNetDraw  : if netOn then begin
                            if NetAsk('/draw') then ;
                         end;
            miNetResgn : if netOn then
                            if CautionAlert(allAlerts, alrtResign) then
                               if NetAsk('/resign') then ;
            miNetAgain : if netOn then begin
                            if NetAsk('/again') then ;
                         end;
            miNetName  : if not netOn then begin
                            viewMd := vwName;
                            nameTxt := '';
                            DrawShow
                         end;
            miNet9600, miNet19200 :
               begin
                  if item = miNet19200 then cnBaud := 19200 else cnBaud := 9600;
                  if CnSave(myVol) then ;
                  if cnBaud = 19200 then status := 'Network speed: 19200 baud.'
                  else status := 'Network speed: 9600 baud.';
                  SyncPort;
                  DrawPanel
               end;
         end;

      mHelp :
         case item of
            miHowTo : NoteAlert(allAlerts, alrtHow);
            miKeys  : NoteAlert(allAlerts, alrtKeys);
         end;

      mVer :
         case item of
            miAbout : NoteAlert(allAlerts, alrtAbout);
         end;

   end
end;

procedure PlainKey (ch : char);
{ Keypad 4 6 8 2 (or J L I K) move the cursor, RETURN / SPACE pick the  }
{ cursor square, N new, U undo, D dots, # numbers, S switch, 1-4 level, }
{ H help.  REV B: on the table list, a digit sits down, R looks again, }
{ N names you, Q goes back; at a table, A asks for another game.        }
begin
   if viewMd = vwName then begin
      NameKey(ch);
      exit(PlainKey)
   end;
   if viewMd = vwTables then begin
      if (ch >= '1') and (ch <= '9') then
         SitAt(ord(ch) - ord('0'))
      else if (ch = 'R') or (ch = 'r') then
         LookTables
      else if (ch = 'N') or (ch = 'n') then begin
         viewMd := vwName;
         nameTxt := '';
         DrawShow
      end
      else if (ch = 'Q') or (ch = 'q') or (ch = chr(27)) then begin
         viewMd := vwBoard;
         DrawShow
      end;
      exit(PlainKey)
   end;
   if netOn and gameOver and ((ch = 'A') or (ch = 'a')) then begin
      if NetAsk('/again') then ;
      exit(PlainKey)
   end;
   if (ch = '4') or (ch = 'J') or (ch = 'j') then
      MoveCursor(0, -1)
   else if (ch = '6') or (ch = 'L') or (ch = 'l') then
      MoveCursor(0, 1)
   else if (ch = '8') or (ch = 'I') or (ch = 'i') then
      MoveCursor(-1, 0)
   else if (ch = '2') or (ch = 'K') or (ch = 'k') then
      MoveCursor(1, 0)
   else if (ch = chr(13)) or (ch = chr(3)) or (ch = ' ') then
      PickSq(ScrSq(cursorR, cursorC))
   else if (ch = 'N') or (ch = 'n') then
      AskNewGame
   else if (ch = 'U') or (ch = 'u') then
      UndoMove
   else if (ch = 'D') or (ch = 'd') then
      ToggleDots
   else if ch = '#' then
      ToggleNums
   else if (ch = 'S') or (ch = 's') then
      SwitchSides
   else if (ch = '1') or (ch = '3') then      { 2 and 4 move the cursor: }
      SetLevel(ord(ch) - ord('0'))           { Apple-2 and Apple-4 (Reversi) }
   else if (ch = 'H') or (ch = 'h') or (ch = '?') then
      NoteAlert(allAlerts, alrtHow)
end;

procedure ProcessTheEvent;
var theMenu, theItem : integer;
begin
   case theEvent.what of

      buttonDown :
         if theEvent.who = menuFolder then begin
            MenuSelect(theEvent.where, theMenu, theItem);
            if (theMenu <> 0) and (theItem <> 0) then
               MenuCommand(theMenu, theItem);
            HiLiteMenu(0)
         end
         else if theEvent.who = myWindow then begin
            ckSeed := (ckSeed + theEvent.where.h * 7 + theEvent.where.v * 13) mod 32749;
            ClickSq(theEvent.where)
         end;

      buttonUp :
         ;

      keyDown :
         if (theEvent.who = myWindow) and (not theEvent.repeatKey) then begin
            ckSeed := (ckSeed + ord(theEvent.ascii) + 31 + idleCnt mod 997) mod 32749;
            if theEvent.appleKey then begin
               MenuKey(theEvent.ascii, theMenu, theItem);
               if theMenu <> 0 then begin
                  HiLiteMenu(theMenu);
                  MenuCommand(theMenu, theItem);
                  HiLiteMenu(0)
               end
            end
            else
               PlainKey(theEvent.ascii)
         end;

      folderActivate :
         begin
            TakeControl(theEvent, false, false);
            PutUpMenus
         end;

      folderDeactivate :
         GiveControl(theEvent);

      folderUpdate :
         if theEvent.who = myWindow then
            doUpdateEvent(true, theEvent.who)
         else begin
            BeginUpdate(theEvent.who);
            EndUpdate(theEvent.who)
         end;

      folderMoved :
         if theEvent.who = myWindow then begin
            SyncPort;
            DrawShow
         end;

      filerEvent :
         DoFilerEvent;

      abortEvent, diedEvent, diskEvent, catalogEvent, nilEvent :
         ;

      otherwise

   end
end;

procedure Initialize;
var err      : integer;
    schedErr : integer;
    pcInfo   : procinfoRec;
    excName  : t_ex_name;
    myPrefix : TParamAlert;
    cutAt    : integer;
    i        : integer;
    tmp1     : string[1];
begin
   initing := true;

   excName := 'SYS_TERMINATE';
   Declare_Excep_Hdl(err, excName, @LastWishes);
   if err <> 0 then begin
      TellFiler(err, initFailed, noMemory, nil);
      halt
   end;

   toldToDie := false;
   myWindow  := nil;
   human     := ckBlack;
   lisaC     := ckWhite;
   level     := 2;
   lvDepth[1] := 2;                   { how far each level looks, and the }
   lvBudget[1] := 400;                { most positions it may take (the   }
   lvDepth[2] := 4;                   { Lisa looks one depth less when a  }
   lvBudget[2] := 1500;               { position is that busy)            }
   lvDepth[3] := 6;
   lvBudget[3] := 4000;
   lvDepth[4] := 8;
   lvBudget[4] := 10000;
   showDots  := true;
   showNums  := false;
   gameOver  := false;
   started   := false;
   thinking  := false;
   nHist     := 0;
   nLegal    := 0;
   nPath     := -1;
   cursorR   := 5;
   cursorC   := 2;
   haveYou   := false;
   haveLisa  := false;
   nodes     := 0;
   status    := '';
   idleCnt   := 0;
   viewMd    := vwBoard;              { REV B }
   netOn     := false;
   lastPoll  := 0;
   lastMvCnt := -1;
   nameTxt   := '';
   CnInit;
   for i := 0 to 7 do                 { 50% gray: 170 / 85 alternating }
      if odd(i) then ckrGray[i] := 85 else ckrGray[i] := 170;
   CkInit;

   InitHeap(@myZone[0], @myZone[maxHeap], @OutOfHeapSpace);

   Sched_Class(schedErr, false);
   Info_Process(err, My_ID, pcInfo);
   Sched_Class(schedErr, true);

   cutAt := pos('}', pcInfo.progpathname);
   if (cutAt = 0) or (cutAt > 40) then begin
      TellFiler(err, initFailed, internalError, nil);
      halt
   end;
   myPrefix := copy(pcInfo.progpathname, 1, cutAt);
   tmp1 := ' ';                         { REV B: the volume, for CKRNAME.DATA }
   tmp1[1] := chr(123);
   cutAt := pos(tmp1, pcInfo.progpathname);
   if (cutAt > 1) and (cutAt <= 33) then
      myVol := copy(pcInfo.progpathname, 1, cutAt - 1)
   else
      myVol := '-#12-';
   if CnLoad(myVol) then ;

   InitMP(theHeap, myPrefix, @AlertFailed, alertVerNum,
          @myMenus[mFile], allAlerts, err);
   if err > 0 then begin
      TellFiler(err, initFailed, noMemory, nil);
      halt
   end;

   InitWmlCrs(err);
   if err <> 0 then begin
      TellFiler(err, initFailed, noDiskSpace, nil);
      halt
   end;

   SetPort(deskPort);

   initing := false
end;

{------------------------------------------------------------------------}
{ The main program -- two-phase loop; the idle branch only counts.       }
{------------------------------------------------------------------------}

begin { main program }

   InitGraf(@thePort);
   OpenWm;
   Initialize;

   repeat

      while not (ImActive or toldToDie) do begin
         GetEvent(theEvent);
         ProcessTheEvent
      end;

      while ImActive and (not toldToDie) do begin
         if EventAvail then begin
            GetEvent(theEvent);
            ProcessTheEvent
         end
         else begin
            idleCnt := idleCnt + 1;
            if idleCnt > 1000000 then idleCnt := 0;
            IdlePoll;                         { REV B }
            LetOthersRun
         end
      end;

   until toldToDie;

   termErr := 0
end.
